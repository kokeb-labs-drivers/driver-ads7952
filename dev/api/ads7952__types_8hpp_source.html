451struct MonitorWindow {
452 uint32_t expected;
453 uint32_t other;
454 uint32_t margin;

457[[nodiscard]] constexpr MonitorWindow WindowOf(const Config& config) noexcept {
458 const Range other = (config.range == Range::TwoVref) ? Range::Vref : Range::TwoVref;
459 const uint32_t fs = FullScaleUv(config.range, config.vref_mv);
460 const uint64_t margin =
461 ((static_cast<uint64_t>(config.monitor.tolerance_uv) * 4096U) + fs - 1U) / fs;
462 return {.expected = CodeOfInput(config.monitor.expected_uv, fs),
463 .other = CodeOfInput(config.monitor.expected_uv, FullScaleUv(other, config.vref_mv)),
464 .margin = (margin > 4096U) ? 4096U : static_cast<uint32_t>(margin)};
465}

480[[nodiscard]] constexpr Error ValidateMonitor(const Config& config) noexcept {
481 const ScaleMonitor& m = config.monitor;
482 if (m.channel == kNoMonitor) {
483 return Error::Ok;
484 }
485 if ((m.channel >= ChannelCount(config.variant)) || (m.tolerance_uv == 0U)) {
486 return Error::InvalidArgument;
487 }
488 const detail::MonitorWindow w = detail::WindowOf(config);
489 if ((w.expected <= w.margin) || ((w.expected + w.margin) >= kMaxCode)) {
490 return Error::InvalidArgument;
491 }
492 const uint32_t gap = (w.expected > w.other) ? (w.expected - w.other) : (w.other - w.expected);
493 return (gap > (2U * w.margin)) ? Error::Ok : Error::InvalidArgument;
494}

504[[nodiscard]] constexpr Error CheckScaleCode(const Config& config, uint16_t code) noexcept {
505 if ((config.monitor.channel == kNoMonitor) || (ValidateMonitor(config) != Error::Ok)) {
506 return Error::InvalidArgument;
507 }
508 const detail::MonitorWindow w = detail::WindowOf(config);
509 const uint32_t c = static_cast<uint32_t>(code) & kMaxCode;
510 if (detail::Within(c, w.expected, w.margin)) {
511 return Error::Ok;
512 }
513 return detail::Within(c, w.other, w.margin) ? Error::RangeMismatch : Error::ScaleMismatch;
514}

521[[nodiscard]] constexpr Error ValidateGpio(const Config& config) noexcept {
522 const GpioConfig& g = config.gpio;
523 const auto alarm = static_cast<uint32_t>(detail::ToUnderlying(g.alarm_output));
524 const uint32_t outputs = g.output_mask;
525 const uint32_t levels = g.output_levels;
526 if ((alarm > 6U) || (alarm == 3U) || (alarm == 5U) || (outputs > 0x0FU) ||
527 ((levels & ~outputs) != 0U)) {
528 return Error::InvalidArgument;
529 }
530 // A pin with a function ignores its direction bit (Table 11 note): it cannot be an output. The
531 // bits collect in uint32_t, so no compound assignment narrows (KB-CPP20-022).
532 uint32_t function_pins = 0U;
533 function_pins |= ((alarm & 0x03U) != 0U) ? 0x01U : 0x00U; // GPIO0: any code with bit 0 or bit 1
534 function_pins |= ((alarm & 0x04U) != 0U) ? 0x02U : 0x00U; // GPIO1: codes 100 and 110
535 function_pins |= g.gpio2_range_input ? 0x04U : 0x00U;
536 function_pins |= g.gpio3_power_down_input ? 0x08U : 0x00U;
537 if ((function_pins & outputs) != 0U) {
538 return Error::InvalidArgument;
539 }
540 if ((config.package == Package::Vqfn) && (((function_pins | outputs) & 0x0EU) != 0U)) {
541 return Error::Unsupported;
542 }
543 return Error::Ok;
544}

552[[nodiscard]] constexpr Error ValidateAlarms(const Config& config) noexcept {
553 const uint8_t channels = ChannelCount(config.variant);
554 for (std::size_t ch = 0; ch < kMaxChannels; ++ch) {
555 const AlarmThreshold& a = config.alarms[ch];
556 const bool reset = (a.low_code < 4U) && (a.high_code >= 4092U);
557 if ((a.low_code > kMaxCode) || (a.high_code > kMaxCode) || ((ch >= channels) && !reset)) {
558 return Error::InvalidArgument;
559 }
560 }
561 return Error::Ok;
562}

598[[nodiscard]] constexpr Error Validate(const Config& config) noexcept {
599 const uint8_t channels = ChannelCount(config.variant);
600 if (!detail::IsEnumerator(config.variant) || !detail::IsEnumerator(config.package) ||
601 !detail::IsEnumerator(config.range)) {
602 return Error::InvalidArgument;
603 }
604 if (!ValidSupplies(config)) {
605 return Error::InvalidArgument;
606 }
607 const auto all = static_cast<uint32_t>(AllChannels(config.variant));
608 if ((config.auto1_channels == 0U) ||
609 ((static_cast<uint32_t>(config.auto1_channels) & ~all) != 0U) ||
610 (config.auto2_last_channel >= channels)) {
611 return Error::InvalidArgument;
612 }
613 if ((ValidateAlarms(config) != Error::Ok) || (ValidateMonitor(config) != Error::Ok)) {
614 return Error::InvalidArgument;
615 }
616 if (!ValidPark(config)) {
617 return Error::InvalidArgument;
618 }
619 return ValidateGpio(config);
620}

628[[nodiscard]] constexpr Config ResetConfig(Variant variant) noexcept {

653[[nodiscard]] constexpr uint32_t CodeToMicrovolts(uint16_t code, Range range,
654 uint16_t vref_mv) noexcept {
655 const uint32_t full_scale_uv = detail::FullScaleUv(range, vref_mv);
656 return static_cast<uint32_t>(
657 (static_cast<uint64_t>(static_cast<uint32_t>(code) & kMaxCode) * full_scale_uv) /
658 kCodesPerFullScale);
659}
