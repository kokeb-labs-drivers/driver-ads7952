140 [[nodiscard]] Error Frame(uint16_t word, uint16_t& reply) noexcept {
141 const uint32_t w = word;
142 tx_ = {static_cast<uint8_t>(w >> 8U), static_cast<uint8_t>(w & 0xFFU)};
143 rx_ = {0xFF, 0xFF};
144 if (!port_.Transfer(tx_.data(), rx_.data(), tx_.size())) {
145 reply = 0xFFFFU;
146 return Error::Bus;
147 }
148 reply = static_cast<uint16_t>((static_cast<uint32_t>(rx_[0]) << 8U) |
149 static_cast<uint32_t>(rx_[1]));
150 return Error::Ok;
151 }
152
153private:
154 Port& port_;
155 std::array<uint8_t, kFrameBytes> tx_{};
156 std::array<uint8_t, kFrameBytes> rx_{};
157};
