156[[nodiscard]] constexpr uint16_t AlarmThresholdWord(uint8_t channel_in_group, bool high, bool exit,
157 uint16_t threshold) noexcept {
158 return static_cast<uint16_t>(

160 (high ? static_cast<uint32_t>(kAlarmHigh) : 0U) |
161 (exit ? static_cast<uint32_t>(kAlarmExit) : 0U) |
