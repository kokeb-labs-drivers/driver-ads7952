34// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
35// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
36// not match (KB-CPP20-007).

39template <typename Port>
40concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(1U); };

44template <typename Port>
45concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };

48template <typename Port>
49concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };

52template <typename Port>
53concept HasLog =
54 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };

83[[nodiscard]] constexpr uint8_t NextInSequence(uint16_t mask, uint8_t channel) noexcept {
84 const uint32_t from = (static_cast<uint32_t>(channel) + 1U) & 0x0FU;
85 const auto rotated = std::rotr(mask, static_cast<int>(from));
86 if (rotated == 0U) {
87 return channel;
88 }
89 const auto step = static_cast<uint32_t>(std::countr_zero(rotated));
90 return static_cast<uint8_t>((from + step) & 0x0FU);
91}

99[[nodiscard]] constexpr uint16_t Auto2Mask(uint8_t last) noexcept {
100 return static_cast<uint16_t>((uint32_t{2} << (static_cast<uint32_t>(last) & 0x0FU)) - 1U);
101}

103// Sections 8.4.5 and 8.4.6: ascending order with wrap-round.
104static_assert((NextInSequence(0x0FFF_u16, 3_u8) == 4U) &&
105 (NextInSequence(0x0FFF_u16, 11_u8) == 0U));
106static_assert((NextInSequence(0x8001_u16, 15_u8) == 0U) &&
107 (NextInSequence(0x8001_u16, 0_u8) == 15U));
108static_assert((NextInSequence(0x0010_u16, 4_u8) == 4U) && (NextInSequence(0x0000_u16, 7_u8) == 7U));
109static_assert((CountChannels(0xFFFF_u16) == 16U) && (CountChannels(0x00A5_u16) == 4U));
110static_assert((Auto2Mask(0_u8) == 0x0001U) && (Auto2Mask(11_u8) == 0x0FFFU) &&

115[[nodiscard]] constexpr uint16_t ControlBits(const Config& config) noexcept {
116 return static_cast<uint16_t>(

123[[nodiscard]] constexpr uint16_t GpioProgramWord(const Config& config) noexcept {
124 const GpioConfig& g = config.gpio;
125 const uint32_t qfn = (config.package == Package::Vqfn) ? 0x000EU : 0x0000U;
126 return static_cast<uint16_t>(
127 static_cast<uint32_t>(reg::kGpioProgram) |
128 (g.gpio3_power_down_input ? static_cast<uint32_t>(reg::kGpio3PowerDown) : 0U) |
129 (g.gpio2_range_input ? static_cast<uint32_t>(reg::kGpio2Range) : 0U) |
130 ((static_cast<uint32_t>(ToUnderlying(g.alarm_output)) & 0x07U) << reg::kGpioAlarmShift) |
131 ((static_cast<uint32_t>(g.output_mask) | qfn) & reg::kGpioDirectionMask));
132}

136[[nodiscard]] constexpr Config EveryThresholdDiffers(const Config& config) noexcept {
137 Config from = config;
138 for (AlarmThreshold& t : from.alarms) {
139 t.low_code = static_cast<uint16_t>(~static_cast<uint32_t>(t.low_code) & kMaxCode);
140 t.high_code = static_cast<uint16_t>(~static_cast<uint32_t>(t.high_code) & kMaxCode);
141 }
142 return from;

156[[nodiscard]] constexpr IdentityChannels IdentityFor(Variant variant) noexcept {
157 switch (variant) {
158 case Variant::Ads7950:
159 return {.first = 2, .second = 1}; // 10, 01
160 case Variant::Ads7951:
161 return {.first = 5, .second = 2}; // 101, 010
162 case Variant::Ads7952:
163 case Variant::Ads7953:
164 default:
165 break;
166 }
167 return {.first = 10, .second = 5}; // 1010, 0101
168}

171[[nodiscard]] constexpr bool IdentityCovers(Variant variant) noexcept {
172 const IdentityChannels ids = IdentityFor(variant);
173 const auto channels = static_cast<uint32_t>(ChannelCount(variant));
174 const uint32_t used = (channels > 8U) ? 0x0FU : (channels - 1U);
175 return (ids.first < channels) && (ids.second < channels) &&
176 (((static_cast<uint32_t>(ids.first) ^ static_cast<uint32_t>(ids.second)) & used) == used);
177}
178
179static_assert(IdentityCovers(Variant::Ads7950) && IdentityCovers(Variant::Ads7951) &&
180 IdentityCovers(Variant::Ads7952) && IdentityCovers(Variant::Ads7953));
